routine identity[T, N](value: T) -> T
needs U64 N
    var tab_parser = CSVParser['\t']
    var tab = '\t'

routine describe(point: (S64, S64)) -> Text
    when point
        (0, 0) => return "origin"
        (x, 0) => return f"x-axis {x}"
        (x, y) => return f"at {x}, {y}"
    unless point.item0 > 5
        return "small"
    else
        return "big"
