module Tests/Format/Layout

import IO/Console
failable routine helper(a: S32) -> S32
    if a < 0
        throw ValueError(message: "negative")
    if not (a != 0)
        throw AbsentValueError()
    return a

routine spacing(xs: List[S32], perms: Perm, v: Shape)
    var total = 0
    var hx: U32 = 0xFF
    var mixed: S32 = (1 + 2) * 3
    var bits = (hx & (1 << 4)) != 0
    var neg = -5
    var call_on_neg = (-5).abs()
    var minus_paren = -(5)
    var r = 0 to 10 by 2
    var inside = (0 to 10).has(5)
    var q = (try helper(a: 1))!! + 1
    var p = try helper(a: 1)!!
    var raw = r"raw\n"
    var bs = b"bytes\x01"
    var ch = 'a'
    var text = "tab\tquote\"backslash\\"
    var f = f"value {hx:=} {{braces}} {neg}"
    var tup = (1,)
    var empty = {:}
    var st = {1, 2}
    var dct = {1: "a", 2: "b"}
    var doubled = xs.select(_ * 2)
    var ranged = xs.where(x given (lo, hi) => lo <= x < hi)
    var add = (x: S32, y: S32) => x + y
    var last = xs[^1]
    var bigger = if total > 3 then total else 3
    for x in xs
        total += x
    then
        show("done")
    for (i, x) in xs.enumerate()
        drop i
    var n = 10
    until n <= 0
        n -= 1
    loop
        break
    then
        show("left")
    var w: U8 = 250
    w +%= 10
    var m: S32? = none
    m ??= 5
    when
        perms.has(READ) => nothing
        perms.lacks(WRITE) => nothing
        else => nothing
    when v
        is Circle as c and c.radius > 0 => show("circle")
        is Rect(w, h) => show("rect")
        isnot Square => nothing
        else => nothing
    when
        v is Circle => show("c")
        else => show("other")
    var access = when
        perms.has(READ and WRITE) => "rw"
        perms.lacks(READ) => "no read"
        else => "other"
    if (v is Circle) and perms.has_any(READ and WRITE) and perms.has(EXEC)
        nothing
    elseif total == 0
        nothing
    else
        nothing
    using open(path: "a") as fa, open(path: "b") as fb
        show("both")
    danger
        var unsafe_value = 1
    return
